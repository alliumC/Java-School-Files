class BankAccount {
   private String accountNumber;
   private double balance;
   
   // Accessor and mutator for Account Number
   public String getAccountNumber() {
      return accountNumber;
   }
   
   public void setAccountNumber(String accountNumber) {
      this.accountNumber = accountNumber;
   }
   
   // Accessor and mutator for balance
   public double getBalance() {
      return balance;
   }
   
   public void setBalance(double balance) {
      if (balance >= 0) {
         this.balance = balance;
      } else {
         System.out.println("Error: Balance must not be a negative number.");
      }
   }
   
   public void deposit(double amount) {
      if (amount > 0) {
         balance += amount;
      } else {
         System.out.println("Error: Deposit amount must not be a negative number.");
      }
   }
   
   public void withdraw(double amount) {
      if (amount > 0 && balance >= amount) {
         balance -= amount;
      } else {
         if (amount<0){
             System.out.println("Error: Withdraw amount must not be a negative number.");
         } else{
             System.out.println("Error: Withdraw amount must not be greater than the account balance.");
         }
      }
   }
}

public class Main {
   public static void main(String[] args) {
      BankAccount bankAccount = new BankAccount();
      
      // Setting Account Values
      bankAccount.setAccountNumber("143143143");
      bankAccount.setBalance(143.00);
      
      // Attempt to set invalid balance
      bankAccount.setBalance(-5);
      
      // Accessing values
      System.out.println("\n|| ACCOUNT DETAILS ||");
      System.out.println("Account Number: " + bankAccount.getAccountNumber());
      System.out.println("Balance: PHP" + bankAccount.getBalance());
      
      // Attempt to set valid and invalid deposit amount
      System.out.println("\n|| DEPOSIT ||");
      bankAccount.deposit(99.00); // valid
      bankAccount.deposit(-99.00); // invalid
      System.out.println("Balance: PHP" + bankAccount.getBalance());
      
      System.out.println("\n|| WITHDRAW ||");
      
      // Attempt to set valid and invalid withdrawal amount
      bankAccount.withdraw(99.00); //valid
      bankAccount.withdraw(500.00); //invalid
      bankAccount.withdraw(-99.00); //invalid
      System.out.println("Balance: PHP" + bankAccount.getBalance());
   }
}
